const toSlug = (value) =>
  slugify(String(value || ''), {
    lower: true,
    strict: true,
    replacement: '-',
    trim: true,
  });

const parseBoolean = (value, fallback = true) => {
  if (typeof value === 'boolean') {
    return value;
  }
  if (typeof value === 'string') {
    if (value.toLowerCase() === 'true') {
      return true;
    }
    if (value.toLowerCase() === 'false') {
      return false;
    }
  }
  return fallback;
};

const parseJsonField = (value, fallbackValue) => {
  if (value === undefined || value === null || value === '') {
    return fallbackValue;
  }

  if (typeof value !== 'string') {
    return value;
  }

  try {
    return JSON.parse(value);
  } catch {
    return fallbackValue;
  }
};

const toNumber = (value, fallback = 0) => {
  const parsed = Number(value);
  return Number.isFinite(parsed) ? parsed : fallback;
};

const normalizeArray = (value) => {
  if (!value) {
    return [];
  }
  if (Array.isArray(value)) {
    return value;
  }
  return [value];
};

const buildUploadedUrls = ({ files, req }) => {
  if (!files) {
    return { primaryImage: null, galleryUrls: [] };
  }

  const toAbsoluteUrl = (filename) => `${req.protocol}://${req.get('host')}${getFileUrl(filename)}`;

  const mainImageFile = files.image?.[0] || null;
  const primaryImage = mainImageFile ? toAbsoluteUrl(mainImageFile.filename) : null;

  const galleryUrls = normalizeArray(files.gallery || [])
    .filter((file) => file?.filename)
    .map((file) => toAbsoluteUrl(file.filename));

  return { primaryImage, galleryUrls };
};

const normalizeSizes = (sizes) => {
  const parsed = parseJsonField(sizes, []);
  if (!Array.isArray(parsed)) {
    return [];
  }

  return parsed
    .map((size) => ({
      label: String(size?.label || '').trim(),
      grams: toNumber(size?.grams, NaN),
      price: toNumber(size?.price, NaN),
      stock: toNumber(size?.stock, NaN),
      value: size?.value ? String(size.value) : null,
    }))
    .filter((size) => size.label);
};

const normalizeBulkPricing = (bulkPricing) => {
  const parsed = parseJsonField(bulkPricing, []);
  if (!Array.isArray(parsed)) {
    return [];
  }

  return parsed
    .map((tier) => ({
      quantity: Math.max(1, Math.trunc(toNumber(tier?.quantity, 1))),
      price: Math.max(0, toNumber(tier?.price, 0)),
      originalPrice: Math.max(0, toNumber(tier?.originalPrice, 0)),
      discount: Math.max(0, toNumber(tier?.discount, 0)),
    }))
    .filter((tier) => tier.quantity > 0);
};

const mergeImageArrays = (...sources) => {
  const unique = [];

  sources.forEach((source) => {
    normalizeArray(source).forEach((value) => {
      if (!value || typeof value !== 'string') {
        return;
      }
      const trimmed = value.trim();
      if (!trimmed || unique.includes(trimmed)) {
        return;
      }
      unique.push(trimmed);
    });
  });

  return unique;
};

const removeProductImages = async (product) => {
  const allImages = mergeImageArrays(product?.image, product?.images, product?.gallery);

  await Promise.all(
    allImages.map(async (imageUrl) => {
      const imageName = imageUrl?.split('/uploads/')[1];
      if (!imageName) {
        return;
      }

      const imagePath = path.join(__dirname, '../../uploads', imageName);
      if (fs.existsSync(imagePath)) {
        await deleteFile(imagePath);
      }
    })
  );
};

const ensureUniqueSlug = async (slug, ignoreId = null) => {
  const query = { slug };
  if (ignoreId) {
    query._id = { $ne: ignoreId };
  }

  const existing = await ProductModel.findOne(query).select('_id').lean();
  if (existing) {
    throw handleError(409, 'Product slug already exists.');
  }
};

const resolveCategory = async (categoryId, fallbackCategoryName = '') => {
  if (!categoryId) {
    throw handleError(400, 'categoryId is required.');
  }

  if (!mongoose.Types.ObjectId.isValid(categoryId)) {
    throw handleError(400, 'Invalid categoryId.');
  }

  const category = await CategoryModel.findById(categoryId).lean();
  if (!category) {
    throw handleError(404, 'Selected category not found.');
  }

  return {
    categoryId: category._id,
    categoryName: fallbackCategoryName || category.name,
  };
};

const buildProductPayload = async ({ req, existingProduct = null }) => {
  const hasField = (fieldName) => Object.prototype.hasOwnProperty.call(req.body, fieldName);

  const payload = {
    name: hasField('name') ? req.body.name : existingProduct?.name,
    slug: req.body.slug ? toSlug(req.body.slug) : undefined,
    categoryId: hasField('categoryId') ? req.body.categoryId : existingProduct?.categoryId || undefined,
    category: hasField('category') ? req.body.category : existingProduct?.category || '',
    variety: hasField('variety') ? req.body.variety || null : existingProduct?.variety || null,
    badge: hasField('badge') ? req.body.badge || null : existingProduct?.badge || null,
    shortDescription: hasField('shortDescription')
      ? req.body.shortDescription || ''
      : existingProduct?.shortDescription || '',
    description: hasField('description') ? req.body.description : existingProduct?.description,
    features: hasField('features')
      ? parseJsonField(req.body.features, [])
      : existingProduct?.features || [],
    sizes: hasField('sizes') ? normalizeSizes(req.body.sizes) : existingProduct?.sizes || [],
    bulkPricing: hasField('bulkPricing')
      ? normalizeBulkPricing(req.body.bulkPricing)
      : existingProduct?.bulkPricing || [],
    rating: hasField('rating') ? toNumber(req.body.rating, 0) : toNumber(existingProduct?.rating, 0),
    reviews: hasField('reviews')
      ? Math.max(0, Math.trunc(toNumber(req.body.reviews, 0)))
      : Math.max(0, Math.trunc(toNumber(existingProduct?.reviews, 0))),
    isActive: parseBoolean(req.body.isActive, existingProduct?.isActive ?? true),
    image: hasField('image') ? req.body.image || '' : existingProduct?.image || '',
    images: hasField('images') ? parseJsonField(req.body.images, []) : existingProduct?.images || [],
    gallery: hasField('gallery') ? parseJsonField(req.body.gallery, []) : existingProduct?.gallery || [],
  };

  if (!payload.slug) {
    if (existingProduct?.slug) {
      payload.slug = existingProduct.slug;
    } else if (payload.name) {
      payload.slug = toSlug(payload.name);
    }
  }

  if (!payload.shortDescription && payload.description) {
    payload.shortDescription = String(payload.description).slice(0, 160);
  }

  if (payload.variety && !mongoose.Types.ObjectId.isValid(payload.variety)) {
    throw handleError(400, 'Invalid variety ID.');
  }

  if (payload.variety) {
    const varietyExists = await VarietyModel.findById(payload.variety).select('_id').lean();
    if (!varietyExists) {
      throw handleError(404, 'Selected variety does not exist.');
    }
  }

  const categoryMeta = await resolveCategory(payload.categoryId, payload.category);
  payload.categoryId = String(categoryMeta.categoryId);
  payload.category = categoryMeta.categoryName;

  const { primaryImage, galleryUrls } = buildUploadedUrls({ files: req.files, req });
  if (primaryImage) {
    payload.image = primaryImage;
  }

  const baseImages = mergeImageArrays(payload.images, payload.gallery, galleryUrls);
  const mergedImages = mergeImageArrays(payload.image, baseImages);

  if (mergedImages.length > 0) {
    payload.images = mergedImages;
    payload.image = mergedImages[0];
  } else if (existingProduct?.images?.length) {
    payload.images = existingProduct.images;
    payload.image = existingProduct.image;
  }

  if (!payload.image) {
    throw handleError(400, 'At least one product image is required.');
  }

  payload.gallery = mergeImageArrays(payload.gallery, galleryUrls).filter((image) => image !== payload.image);

  const lowestSizePrice =
    payload.sizes.length > 0 ? Math.min(...payload.sizes.map((size) => Number(size.price) || 0)) : null;

  payload.price = lowestSizePrice !== null ? lowestSizePrice : toNumber(req.body.price, existingProduct?.price || 0);
  payload.originalPrice = Math.max(
    payload.price,
    toNumber(req.body.originalPrice, existingProduct?.originalPrice || payload.price)
  );

  if (!validateSizeLabels(payload.sizes)) {
    throw handleError(400, 'Size labels must be unique per product.');
  }

  return payload;
};

const cleanForUpdate = (payload) => {
  const clean = { ...payload };
  delete clean.createdAt;
  delete clean.updatedAt;
  delete clean.__v;
  return clean;
};

// Create Product
export const createProduct = async (req, res, next) => {
  try {
    const payload = await buildProductPayload({ req });

    const { error } = productCreateValidationSchema.validate(payload, {
      abortEarly: false,
      stripUnknown: true,
    });
    if (error) {
      return next(handleError(400, error.details.map((detail) => detail.message).join(' ')));
    }

    await ensureUniqueSlug(payload.slug);

    const createdProduct = await ProductModel.create(payload);
    const populated = await ProductModel.findById(createdProduct._id).populate('categoryId variety');

    return handleSuccess(res, 201, 'Product created successfully.', populated);
  } catch (error) {
    if (error?.code === 11000) {
      return next(handleError(409, 'Product slug already exists.'));
    }
    return next(error);
  }
};

// Update Product
export const updateProduct = async (req, res, next) => {
  try {
    const { id } = req.params;

    if (!mongoose.Types.ObjectId.isValid(id)) {
      return next(handleError(400, 'Invalid product ID.'));
    }

    const existingProduct = await ProductModel.findById(id);
    if (!existingProduct) {
      return next(handleError(404, 'Product not found.'));
    }

    const payload = await buildProductPayload({ req, existingProduct });
    const sanitizedPayload = cleanForUpdate(payload);

    const { error } = productUpdateValidationSchema.validate(sanitizedPayload, {
      abortEarly: false,
      stripUnknown: true,
    });
    if (error) {
      return next(handleError(400, error.details.map((detail) => detail.message).join(' ')));
    }

    await ensureUniqueSlug(sanitizedPayload.slug, id);

    const previousImages = mergeImageArrays(existingProduct.image, existingProduct.images, existingProduct.gallery);
    const updatedImages = mergeImageArrays(
      sanitizedPayload.image,
      sanitizedPayload.images,
      sanitizedPayload.gallery
    );

    const staleImages = previousImages.filter((image) => !updatedImages.includes(image));
    await Promise.all(
      staleImages.map(async (imageUrl) => {
        const imageName = imageUrl?.split('/uploads/')[1];
        if (!imageName) {
          return;
        }
        const imagePath = path.join(__dirname, '../../uploads', imageName);
        if (fs.existsSync(imagePath)) {
          await deleteFile(imagePath);
        }
      })
    );

    const updatedProduct = await ProductModel.findByIdAndUpdate(id, sanitizedPayload, {
      new: true,
      runValidators: true,
    }).populate('categoryId variety');

    if (!updatedProduct) {
      return next(handleError(404, 'Product not found.'));
    }

    return handleSuccess(res, 200, 'Product updated successfully.', updatedProduct);
  } catch (error) {
    if (error?.code === 11000) {
      return next(handleError(409, 'Product slug already exists.'));
    }
    return next(error);
  }
};

// Delete Product
export const deleteProduct = async (req, res, next) => {
  try {
    const { id } = req.params;

    if (!mongoose.Types.ObjectId.isValid(id)) {
      return next(handleError(400, 'Invalid product ID.'));
    }

    const deletedProduct = await ProductModel.findByIdAndDelete(id);
    if (!deletedProduct) {
      return next(handleError(404, 'Product not found.'));
    }

    await removeProductImages(deletedProduct);

    return handleSuccess(res, 200, 'Product deleted successfully.');
  } catch (error) {
    return next(error);
  }
};

const PaymentForm = ({
  onSuccess,
  onFailure,
}: {
  onSuccess: (paymentIntentId: string) => Promise<void>;
  onFailure: (message: string) => void;
}) => {
  const stripe = useStripe();
  const elements = useElements();
  const [isSubmitting, setIsSubmitting] = useState(false);
  const [paymentMessage, setPaymentMessage] = useState<string | null>(null);

  const handleSubmit = async (event: React.FormEvent<HTMLFormElement>) => {
    event.preventDefault();

    if (!stripe || !elements) {
      return;
    }

    setIsSubmitting(true);
    setPaymentMessage(null);

    const returnUrl = typeof window !== 'undefined' ? `${window.location.origin}/checkout` : undefined;

    const { error, paymentIntent } = await stripe.confirmPayment({
      elements,
      confirmParams: returnUrl ? { return_url: returnUrl } : undefined,
      redirect: 'if_required',
    });

    if (error) {
      const message = error.message || 'Payment failed. Please verify your card details.';
      setPaymentMessage(message);
      onFailure(message);
      setIsSubmitting(false);
      return;
    }

    if (!paymentIntent || paymentIntent.status !== 'succeeded') {
      const message = paymentIntent ? `Payment status is ${paymentIntent.status}.` : 'Payment confirmation failed.';
      setPaymentMessage(message);
      onFailure(message);
      setIsSubmitting(false);
      return;
    }

    try {
      await onSuccess(paymentIntent.id);
      setPaymentMessage('Payment completed successfully.');
    } catch (requestError) {
      const message =
        requestError instanceof Error ? requestError.message : 'Payment completed, but order creation failed.';
      setPaymentMessage(message);
      onFailure(message);
    } finally {
      setIsSubmitting(false);
    }
  };

            </div>

  let inThrottle: boolean;
  return function (this: any, ...args: any[]) {

};
