productSchema.pre('validate', function (next) {
  if (this.name && !this.slug) {
    this.slug = slugify(this.name, {
      lower: true,
      strict: true,
      replacement: '-',
      trim: true,
    });
  }

  const uniqueImages = [];
  const addImage = (value) => {
    if (!value || typeof value !== 'string') {
      return;
    }
    const trimmed = value.trim();
    if (!trimmed || uniqueImages.includes(trimmed)) {
      return;
    }
    uniqueImages.push(trimmed);
  };

  addImage(this.image);
  (this.images || []).forEach(addImage);
  (this.gallery || []).forEach(addImage);

  if (uniqueImages.length > 0) {
    this.images = uniqueImages;
    this.image = uniqueImages[0];
  }

  if ((!this.price || this.price < 0) && Array.isArray(this.sizes) && this.sizes.length > 0) {
    this.price = Math.min(...this.sizes.map((size) => Number(size.price) || 0));
  }

  if (!this.originalPrice || this.originalPrice < 0) {
    this.originalPrice = this.price;
  }

  if (!this.shortDescription && this.description) {
    this.shortDescription = this.description.slice(0, 160);
  }

  next();
});

const ProductModel = mongoose.model('Product', productSchema);

export default ProductModel;

import slugify from 'slugify';
import path from 'path';
import fs from 'fs';
import mongoose from 'mongoose';

import CategoryModel from '../models/categoryModel.js';
import ProductModel from '../models/productModel.js';
import handleError from '../utils/errorHandler.js';
import handleSuccess from '../utils/sucessHandler.js';
import { categoryCreateValidationSchema, categoryUpdateValidationSchema } from '../validations/categoryValidationSchema.js';
import { getFileUrl } from '../middlewares/MulterMiddleware/multerMiddleware.js';
import { __dirname, deleteFile } from '../utils/fileHelpers.js';

const toSlug = (value) =>
  slugify(String(value || ''), {
    lower: true,
    strict: true,
    replacement: '-',
    trim: true,
  });

const parseBoolean = (value, fallback = true) => {
  if (typeof value === 'boolean') {
    return value;
  }
  if (typeof value === 'string') {
    if (value.toLowerCase() === 'true') {
      return true;
    }
    if (value.toLowerCase() === 'false') {
      return false;
    }
  }
  return fallback;
};

const buildCategoryPayload = (reqBody = {}, reqFile = null, reqMeta = null, existingCategory = null) => {
  const hasField = (fieldName) => Object.prototype.hasOwnProperty.call(reqBody, fieldName);

  const payload = {
    name: hasField('name') ? reqBody.name : existingCategory?.name,
    slug: hasField('slug') ? toSlug(reqBody.slug) : existingCategory?.slug,
    description: hasField('description') ? reqBody.description || '' : existingCategory?.description || '',
    image: hasField('image') ? reqBody.image || '' : existingCategory?.image || '',
    isActive: hasField('isActive')
      ? parseBoolean(reqBody.isActive, true)
      : (existingCategory?.isActive ?? true),
  };

  if (!payload.slug && payload.name) {
    payload.slug = toSlug(payload.name);
  }

  if (reqFile && reqMeta) {
    const relativeUrl = getFileUrl(reqFile.filename);
    payload.image = `${reqMeta.protocol}://${reqMeta.host}${relativeUrl}`;
  }

  return payload;
};

const ensureUniqueCategory = async ({ slug, name, ignoreId = null }) => {
  const conflictQuery = {
    $or: [{ slug }, { name }],
  };

  if (ignoreId) {
    conflictQuery._id = { $ne: ignoreId };
  }

  const existing = await CategoryModel.findOne(conflictQuery).lean();
  if (!existing) {
    return;
  }

  if (existing.slug === slug) {
    throw handleError(409, 'Category slug already exists.');
  }

  throw handleError(409, 'Category name already exists.');
};

const removeCategoryImageIfExists = async (imageUrl) => {
  const imageName = imageUrl?.split('/uploads/')[1];
  if (!imageName) {
    return;
  }

  const imagePath = path.join(__dirname, '../../uploads', imageName);
  if (fs.existsSync(imagePath)) {
    await deleteFile(imagePath);
  }
};

export const getAllCategories = async (req, res, next) => {
  try {
    const includeInactive = req.query.includeInactive === 'true';
    const filter = includeInactive ? {} : { isActive: true };

    const categories = await CategoryModel.find(filter).sort({ createdAt: 1 }).lean();

    const categoriesWithCount = await Promise.all(
      categories.map(async (category) => {
        const productFilter = { categoryId: category._id };
        if (!includeInactive) {
          productFilter.isActive = true;
        }
        const productCount = await ProductModel.countDocuments(productFilter);
        return { ...category, productCount };
      })
    );

    return handleSuccess(res, 200, 'Categories fetched successfully.', categoriesWithCount);
  } catch (error) {
    return next(error);
  }
};

export const getCategoryBySlug = async (req, res, next) => {
  try {
    const { slug } = req.params;
    const category = await CategoryModel.findOne({ slug: toSlug(slug), isActive: true }).lean();

    if (!category) {
      return next(handleError(404, 'Category not found.'));
    }

    const products = await ProductModel.find({
      categoryId: category._id,
      isActive: true,
    })
      .sort({ createdAt: -1 })
      .lean();

    return handleSuccess(res, 200, 'Category fetched successfully.', {
      category,
      products,
    });
  } catch (error) {
    return next(error);
  }
};

export const createCategory = async (req, res, next) => {
  try {
    const payload = buildCategoryPayload(req.body, req.file, {
      protocol: req.protocol,
      host: req.get('host'),
    });

    const { error } = categoryCreateValidationSchema.validate(payload, { abortEarly: false });
    if (error) {
      return next(handleError(400, error.details.map((detail) => detail.message).join(' ')));
    }

    await ensureUniqueCategory({ slug: payload.slug, name: payload.name });

    const createdCategory = await CategoryModel.create(payload);

    return handleSuccess(res, 201, 'Category created successfully.', createdCategory);
  } catch (error) {
    if (error?.code === 11000) {
      return next(handleError(409, 'Category slug or name already exists.'));
    }
    return next(error);
  }
};

export const updateCategory = async (req, res, next) => {
  try {
    const { id } = req.params;
    if (!mongoose.Types.ObjectId.isValid(id)) {
      return next(handleError(400, 'Invalid category ID.'));
    }

    const existingCategory = await CategoryModel.findById(id);
    if (!existingCategory) {
      return next(handleError(404, 'Category not found.'));
    }

    const payload = buildCategoryPayload(
      req.body,
      req.file,
      {
        protocol: req.protocol,
        host: req.get('host'),
      },
      existingCategory
    );

    const { error } = categoryUpdateValidationSchema.validate(payload, {
      abortEarly: false,
      stripUnknown: true,
    });
    if (error) {
      return next(handleError(400, error.details.map((detail) => detail.message).join(' ')));
    }

    if (!payload.slug && payload.name) {
      payload.slug = toSlug(payload.name);
    }

    if (payload.slug || payload.name) {
      await ensureUniqueCategory({
        slug: payload.slug || existingCategory.slug,
        name: payload.name || existingCategory.name,
        ignoreId: id,
      });
    }

    if (payload.image && existingCategory.image && payload.image !== existingCategory.image) {
      await removeCategoryImageIfExists(existingCategory.image);

const toSlug = (value) =>
  slugify(String(value || ''), {
    lower: true,
    strict: true,
    replacement: '-',
    trim: true,
  });

const parseBoolean = (value, fallback = true) => {
  if (typeof value === 'boolean') {
    return value;
  }
  if (typeof value === 'string') {
    if (value.toLowerCase() === 'true') {
      return true;
    }
    if (value.toLowerCase() === 'false') {
      return false;
    }
  }
  return fallback;
};

const parseJsonField = (value, fallbackValue) => {
  if (value === undefined || value === null || value === '') {
    return fallbackValue;
  }

  if (typeof value !== 'string') {
    return value;
  }

  try {
    return JSON.parse(value);
  } catch {
    return fallbackValue;
  }
};

const toNumber = (value, fallback = 0) => {
  const parsed = Number(value);
  return Number.isFinite(parsed) ? parsed : fallback;
};

const normalizeArray = (value) => {
  if (!value) {
    return [];
  }
  if (Array.isArray(value)) {
    return value;
  }
  return [value];
};

const buildUploadedUrls = ({ files, req }) => {
  if (!files) {
    return { primaryImage: null, galleryUrls: [] };
  }

  const toAbsoluteUrl = (filename) => `${req.protocol}://${req.get('host')}${getFileUrl(filename)}`;

  const mainImageFile = files.image?.[0] || null;
  const primaryImage = mainImageFile ? toAbsoluteUrl(mainImageFile.filename) : null;

  const galleryUrls = normalizeArray(files.gallery || [])
    .filter((file) => file?.filename)
    .map((file) => toAbsoluteUrl(file.filename));

  return { primaryImage, galleryUrls };
};

const normalizeSizes = (sizes) => {
  const parsed = parseJsonField(sizes, []);
  if (!Array.isArray(parsed)) {
    return [];
  }

  return parsed
    .map((size) => ({
      label: String(size?.label || '').trim(),
      grams: toNumber(size?.grams, NaN),
      price: toNumber(size?.price, NaN),
      stock: toNumber(size?.stock, NaN),
      value: size?.value ? String(size.value) : null,
    }))
    .filter((size) => size.label);
};

const normalizeBulkPricing = (bulkPricing) => {
  const parsed = parseJsonField(bulkPricing, []);
  if (!Array.isArray(parsed)) {
    return [];
  }

  return parsed
    .map((tier) => ({
      quantity: Math.max(1, Math.trunc(toNumber(tier?.quantity, 1))),
      price: Math.max(0, toNumber(tier?.price, 0)),
      originalPrice: Math.max(0, toNumber(tier?.originalPrice, 0)),
      discount: Math.max(0, toNumber(tier?.discount, 0)),
    }))
    .filter((tier) => tier.quantity > 0);
};

const mergeImageArrays = (...sources) => {
  const unique = [];

  sources.forEach((source) => {
    normalizeArray(source).forEach((value) => {
      if (!value || typeof value !== 'string') {
        return;
      }
      const trimmed = value.trim();
      if (!trimmed || unique.includes(trimmed)) {
        return;
      }
      unique.push(trimmed);
    });
  });

  return unique;
};

const buildProductPayload = async ({ req, existingProduct = null }) => {
  const hasField = (fieldName) => Object.prototype.hasOwnProperty.call(req.body, fieldName);

  const payload = {
    name: hasField('name') ? req.body.name : existingProduct?.name,
    slug: req.body.slug ? toSlug(req.body.slug) : undefined,
    categoryId: hasField('categoryId') ? req.body.categoryId : existingProduct?.categoryId || undefined,
    category: hasField('category') ? req.body.category : existingProduct?.category || '',
    variety: hasField('variety') ? req.body.variety || null : existingProduct?.variety || null,
    badge: hasField('badge') ? req.body.badge || null : existingProduct?.badge || null,
    shortDescription: hasField('shortDescription')
      ? req.body.shortDescription || ''
      : existingProduct?.shortDescription || '',
    description: hasField('description') ? req.body.description : existingProduct?.description,
    features: hasField('features')
      ? parseJsonField(req.body.features, [])
      : existingProduct?.features || [],
    sizes: hasField('sizes') ? normalizeSizes(req.body.sizes) : existingProduct?.sizes || [],
    bulkPricing: hasField('bulkPricing')
      ? normalizeBulkPricing(req.body.bulkPricing)
      : existingProduct?.bulkPricing || [],
    rating: hasField('rating') ? toNumber(req.body.rating, 0) : toNumber(existingProduct?.rating, 0),
    reviews: hasField('reviews')
      ? Math.max(0, Math.trunc(toNumber(req.body.reviews, 0)))
      : Math.max(0, Math.trunc(toNumber(existingProduct?.reviews, 0))),
    isActive: parseBoolean(req.body.isActive, existingProduct?.isActive ?? true),
    image: hasField('image') ? req.body.image || '' : existingProduct?.image || '',
    images: hasField('images') ? parseJsonField(req.body.images, []) : existingProduct?.images || [],
    gallery: hasField('gallery') ? parseJsonField(req.body.gallery, []) : existingProduct?.gallery || [],
  };

  if (!payload.slug) {
    if (existingProduct?.slug) {
      payload.slug = existingProduct.slug;
    } else if (payload.name) {
      payload.slug = toSlug(payload.name);
    }
  }

  if (!payload.shortDescription && payload.description) {
    payload.shortDescription = String(payload.description).slice(0, 160);
  }

  if (payload.variety && !mongoose.Types.ObjectId.isValid(payload.variety)) {
    throw handleError(400, 'Invalid variety ID.');
  }

  if (payload.variety) {
    const varietyExists = await VarietyModel.findById(payload.variety).select('_id').lean();
    if (!varietyExists) {
      throw handleError(404, 'Selected variety does not exist.');
    }
  }

  const categoryMeta = await resolveCategory(payload.categoryId, payload.category);
  payload.categoryId = String(categoryMeta.categoryId);
  payload.category = categoryMeta.categoryName;

  const { primaryImage, galleryUrls } = buildUploadedUrls({ files: req.files, req });
  if (primaryImage) {
    payload.image = primaryImage;
  }

  const baseImages = mergeImageArrays(payload.images, payload.gallery, galleryUrls);
  const mergedImages = mergeImageArrays(payload.image, baseImages);

  if (mergedImages.length > 0) {
    payload.images = mergedImages;
    payload.image = mergedImages[0];
  } else if (existingProduct?.images?.length) {
    payload.images = existingProduct.images;
    payload.image = existingProduct.image;
  }

  if (!payload.image) {
    throw handleError(400, 'At least one product image is required.');
  }

  payload.gallery = mergeImageArrays(payload.gallery, galleryUrls).filter((image) => image !== payload.image);

  const lowestSizePrice =
